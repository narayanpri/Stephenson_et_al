#-------------------------------------------------------------------------------
# Perform ANOVA
#-------------------------------------------------------------------------------

running.pvs <- NULL
for (i in 1:dim(Final.working.data)[1]) {
	print(i)
	probe.data <- as.numeric(unlist(Final.working.data[i,]))
	temp <- data.frame(DATA=probe.data,GROUP=factor(sample.info$Group))
	aov.results <- aov(DATA~GROUP,data=temp)
	if (deviance(aov.results ) > sqrt(.Machine$double.eps)) {
		temp <- Anova(aov.results,type="III",singular.ok=TRUE)
		temp <- summary(aov.results)
		temp <- as.numeric(unlist(temp))[9]
		running.pvs <- rbind(running.pvs,temp)
	} else {
		running.pvs <- rbind(running.pvs,1)
	}
}
names(running.pvs) <- dimnames(Final.working.data)[[1]]
running.pvs.out <- cbind(running.pvs,running.pvs)
dimnames(running.pvs.out)[[1]] <- dimnames(Final.working.data)[[1]]
write.table(running.pvs.out,"ANOVA_Results.txt",sep="\t",row.names=T)

#-------------------------------------------------------------------------------
# FDR MCC (alpha=0.05)
#-------------------------------------------------------------------------------

temp1 <- mt.rawp2adjp(running.pvs, proc="BH")

dimnames(temp2)[[1]] <- names(running.pvs)
mcc.results <- temp2
write.table(mcc.results,"MCC_Results.txt",sep="\t")

passing.mcc.frags <- dimnames(mcc.results[mcc.results[,2]<0.05,])[[1]]

#-------------------------------------------------------------------------------

#-------------------------------------------------------------------------------

running.diffs <- NULL

	print(i)
	print(dim(posthoc.data)[1])
	temp1 <- as.numeric(posthoc.data[i,])
	temp2 <- data.frame(DATA=temp1,Class=factor(sample.info$Group))
	temp3 <- aov(DATA~Class,data=temp2)
	temp4 <- TukeyHSD(temp3)
	temp5 <- temp4[[1]]
	temp6 <- temp5[,4]
	running.posthoc <- rbind(running.posthoc,temp6)
	temp7 <- temp5[,1]
	running.diffs <- rbind(running.diffs,temp7)
}
running.posthoc <- as.data.frame(running.posthoc)
dimnames(running.posthoc)[[1]] <- dimnames(posthoc.data)[[1]]
coded.running.posthoc <- ifelse(running.posthoc<0.05,1,0)
running.diffs <- as.data.frame(running.diffs)
dimnames(running.diffs)[[1]] <- dimnames(posthoc.data)[[1]]
running.diffs <- as.matrix(running.diffs)
linear.running.diffs <- 2^running.diffs
linear.running.diffs <- ifelse(linear.running.diffs<1,-1/linear.running.diffs,linear.running.diffs)
coded.linear.running.diffs <- ifelse(abs(linear.running.diffs)>=1.5,1,0)

cx <- coded.linear.running.diffs+coded.running.posthoc

keepers <- dimnames(coded.linear.running.diffs)[[1]][apply(cx,1,max)>0]
